cmake_minimum_required(VERSION 3.16)
project(rm_zlio_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(ros_interfaces REQUIRED)
find_package(rm_nav_interfaces REQUIRED)
add_executable(imu_lpf src/imu_lpf.cpp)
target_link_libraries(imu_lpf rclcpp::rclcpp ${sensor_msgs_TARGETS})
target_include_directories(imu_lpf PRIVATE ${sensor_msgs_INCLUDE_DIRS})

add_executable(imu_prep src/imu_prep.cpp)
target_link_libraries(imu_prep rclcpp::rclcpp ${sensor_msgs_TARGETS})
target_include_directories(imu_prep PRIVATE ${sensor_msgs_INCLUDE_DIRS})

add_executable(world_vel_to_body src/world_vel_to_body.cpp)
target_link_libraries(world_vel_to_body rclcpp::rclcpp ${geometry_msgs_TARGETS} ${nav_msgs_TARGETS})
target_include_directories(world_vel_to_body PRIVATE ${geometry_msgs_INCLUDE_DIRS} ${nav_msgs_INCLUDE_DIRS})
target_link_libraries(world_vel_to_body tf2::tf2 tf2_ros::tf2_ros ${tf2_geometry_msgs_TARGETS})

add_executable(vel_diag src/vel_diag.cpp)
target_link_libraries(vel_diag rclcpp::rclcpp ${geometry_msgs_TARGETS} ${nav_msgs_TARGETS} ${ros_interfaces_TARGETS} ${rm_nav_interfaces_TARGETS})
target_include_directories(vel_diag PRIVATE ${geometry_msgs_INCLUDE_DIRS} ${nav_msgs_INCLUDE_DIRS} ${ros_interfaces_INCLUDE_DIRS})

add_executable(zlio_tf_adapter src/zlio_tf_adapter.cpp)
target_link_libraries(zlio_tf_adapter rclcpp::rclcpp ${nav_msgs_TARGETS} tf2::tf2 tf2_ros::tf2_ros)
target_include_directories(zlio_tf_adapter PRIVATE ${nav_msgs_INCLUDE_DIRS})

install(TARGETS zlio_tf_adapter RUNTIME DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/nav_lifecycle_guard.py DESTINATION lib/${PROJECT_NAME})
install(TARGETS vel_diag RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS imu_prep RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS world_vel_to_body RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS imu_lpf RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
